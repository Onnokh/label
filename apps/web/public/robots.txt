# Content Signals state what a crawler may do with what it fetches, separately
# from whether it may fetch it at all. Sleevy wants its docs and marketing pages
# to be findable and to be quotable in an answer with a link back, but it does
# not offer them as model training data.
Content-Signal: search=yes, ai-input=yes, ai-train=no

User-agent: *
Allow: /
# /inbox is intentionally crawlable: it's already indexed from a past crawl and
# is kept out via the X-Robots-Tag: noindex header (server.ts) instead, so
# Google can actually see the noindex signal and drop it. Disallowing it here
# would just block the crawl that's needed to deindex it.
Disallow: /library
Disallow: /settings
Disallow: /connect
Disallow: /oauth/
